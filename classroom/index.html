เปิดสมการและ reward อีก 6 รายการ

r(t) = 0.02 × Σ wᵢ fᵢ(t) ทุก term ประเมินที่ control step เดียวกัน การ reach → grasp → lift เป็นลำดับพฤติกรรมที่สังเกตเห็น

  • reaching = exp(−(ระยะมือถึงลูก / 0.05)²)
  • grasp = exp(−(ระยะมือถึงลูก / 0.025)²) × สัดส่วนการปิด gripper
  • lift = ความสูงเหนือ 0.018 m หารด้วยช่วงถึงเป้า จำกัดใน [0,1]
  • bringing = exp(−(ระยะลูกถึงเป้า / 0.08)²) เมื่อ z > 0.018 m
  • success = 1 เมื่อ z > 0.04 m และระยะลูกถึงเป้า < 0.05 m
  • arm_nominal_pose และ ee_nominal_orientation ให้คะแนนใกล้ท่า hover และมุมมืออ้างอิง น้ำหนักอย่างละ 0.05
  • action_rate, joint_vel_hinge, joint_pos_limits และ floor_contact หักคะแนนเมื่อ action เปลี่ยนเร็ว ข้อหมุนเร็วเกิน ข้อเข้าใกล้ขีดจำกัด หรือชนโต๊ะ ตามลำดับ

B · ค้างลูกให้สูง

จุดแรก เพิ่มฟังก์ชันนี้ถัดจาก reward_cube_lift ใน mujoco_playground/tasks/manipulation/pick_cube/mdp.py โดย z วัดจากพิกัดโลกเป็นเมตร

def reward_cube_high_hold(features, cube_xpos_key="object/xpos",
                          table_z=0.018, z_cap=0.18,
                          hold_bonus_weight=0.5):
  cube_z = features[cube_xpos_key][..., 2]
  height_score = ((cube_z - table_z) / (z_cap - table_z)).clamp(0, 1)
  hold_bonus = (height_score >= 0.9).float()
  return height_score + hold_bonus_weight * hold_bonus

จุดที่สอง เปลี่ยนเฉพาะ factory ของ state-based task ใน so101/task.py ฟังก์ชันใหม่ใช้ slot ชื่อ lift เดิม และปิดคะแนนจากการย้ายไปยังเป้าเดิม

def so101_pick_cube_env() -> config.Environment:
  cfg = pick_task.pick_cube_env()
  _configure_so101_robot(cfg)
  rewards = cfg.task_layer.rewards
  rewards["lift"] = config.Reward(mdp.reward_cube_high_hold, weight=3.0)
  rewards["bringing"].weight = 0.0
  rewards["success"].weight = 0.0
  return cfg
